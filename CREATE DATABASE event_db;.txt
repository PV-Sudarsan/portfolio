#!/bin/bash

echo "Enter number of terms:"
read n

a=0
b=1

echo "Fibonacci Series:"
echo -n "$a $b "

for (( i=2; i<n; i++ ))
do
    c=$((a + b))
    echo -n "$c "
    a=$b
    b=$c
done

echo






#include <stdio.h>

int main() {
    int pages, frames, i, j, page[50], frame[10], flag = 0, page_faults = 0;

    printf("Enter number of pages: ");
    scanf("%d", &pages);

    printf("Enter page reference string:\n");
    for(i = 0; i < pages; i++) {
        scanf("%d", &page[i]);
    }

    printf("Enter number of frames: ");
    scanf("%d", &frames);

    for(i = 0; i < frames; i++) {
        frame[i] = -1;
    }

    for(i = 0; i < pages; i++) {
        flag = 0;

        for(j = 0; j < frames; j++) {
            if(frame[j] == page[i]) {
                flag = 1;
                break;
            }
        }

        if(flag == 0) {
            frame[i % frames] = page[i];
            page_faults++;
        }

        printf("\nFrame status: ");
        for(j = 0; j < frames; j++) {
            printf("%d ", frame[j]);
        }
    }

    printf("\nTotal Page Faults = %d\n", page_faults);

    return 0;
}
